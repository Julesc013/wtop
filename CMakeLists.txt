cmake_minimum_required(VERSION 3.16)
project(Wtop LANGUAGES C)
include(CTest)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" WTOP_VERSION LIMIT_COUNT 1)
configure_file(source/version.h.in generated/version.h @ONLY)
add_library(wtop_core STATIC source/core.c source/settings.c source/ui.c source/text.c source/metrics.c source/actions.c)
target_include_directories(wtop_core PUBLIC source)
set_target_properties(wtop_core PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(BUILD_TESTING)
    foreach(component IN ITEMS core settings ui text metrics actions)
        add_executable(test_${component} tests/test_${component}.c)
        target_link_libraries(test_${component} PRIVATE wtop_core)
        set_target_properties(test_${component} PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if(MSVC)
            target_compile_options(test_${component} PRIVATE /UNDEBUG)
        else()
            target_compile_options(test_${component} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME portable_${component} COMMAND test_${component})
    endforeach()
    add_executable(test_legacy_probe tests/test_legacy_probe.c source/legacy/probe.c)
    target_link_libraries(test_legacy_probe PRIVATE wtop_core)
    set_target_properties(test_legacy_probe PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(test_legacy_probe PRIVATE /UNDEBUG)
        set_property(TARGET test_legacy_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(test_legacy_probe PRIVATE -UNDEBUG)
    endif()
    add_test(NAME portable_legacy_probe COMMAND test_legacy_probe)
endif()
if(WIN32)
    add_executable(wtop source/ntop.c source/util.c source/vi.c source/settings_store_nt.c source/providers_nt.c source/output_nt.c)
    target_include_directories(wtop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(wtop PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 PSAPI_VERSION=1)
    target_link_libraries(wtop PRIVATE wtop_core advapi32 user32 psapi wbemuuid)
    if(MSVC)
        target_compile_options(wtop PRIVATE /W4)
        set_property(TARGET wtop wtop_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        if(BUILD_TESTING)
            set_property(TARGET test_core test_settings test_ui test_text test_metrics test_actions PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
    elseif(MINGW)
        target_link_options(wtop PRIVATE -municode)
        target_link_libraries(wtop PRIVATE m)
    endif()
    if(BUILD_TESTING)
        add_executable(test_settings_store_nt tests/test_settings_store_nt.c source/settings_store_nt.c)
        target_link_libraries(test_settings_store_nt PRIVATE wtop_core advapi32)
        target_compile_definitions(test_settings_store_nt PRIVATE UNICODE _UNICODE WTOP_STORE_TESTING WINVER=0x0601 _WIN32_WINNT=0x0601)
        if(MSVC)
            target_compile_options(test_settings_store_nt PRIVATE /W4 /UNDEBUG)
            set_property(TARGET test_settings_store_nt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        else()
            target_compile_options(test_settings_store_nt PRIVATE -UNDEBUG)
        endif()
        add_test(NAME native_settings_store COMMAND test_settings_store_nt)
        set_tests_properties(native_settings_store PROPERTIES TIMEOUT 30)
        add_executable(test_providers_nt tests/test_providers_nt.c source/providers_nt.c)
        target_compile_definitions(test_providers_nt PRIVATE UNICODE _UNICODE WTOP_PROVIDER_TESTING WINVER=0x0601 _WIN32_WINNT=0x0601)
        target_include_directories(test_providers_nt PRIVATE source)
        target_link_libraries(test_providers_nt PRIVATE advapi32 wbemuuid)
        add_executable(test_output_nt tests/test_output_nt.c source/output_nt.c)
        target_compile_definitions(test_output_nt PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601)
        target_include_directories(test_output_nt PRIVATE source)
        # Compile the real native tree routines into their fixture with only the
        # application entry point renamed; no copied tree implementation.
        add_executable(test_tree_nt tests/test_native_tree.c source/util.c source/vi.c source/settings_store_nt.c source/providers_nt.c source/output_nt.c)
        target_compile_definitions(test_tree_nt PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 PSAPI_VERSION=1)
        target_include_directories(test_tree_nt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
        target_link_libraries(test_tree_nt PRIVATE wtop_core advapi32 user32 psapi wbemuuid)
        foreach(native_component IN ITEMS providers output tree)
            if(MSVC)
                target_compile_options(test_${native_component}_nt PRIVATE /W4 /UNDEBUG)
                set_property(TARGET test_${native_component}_nt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            else()
                target_compile_options(test_${native_component}_nt PRIVATE -UNDEBUG)
            endif()
            add_test(NAME native_${native_component} COMMAND test_${native_component}_nt)
            set_tests_properties(native_${native_component} PROPERTIES TIMEOUT 45)
        endforeach()
        if(MINGW)
            target_link_options(test_providers_nt PRIVATE -municode)
            target_link_libraries(test_tree_nt PRIVATE m)
        endif()
        add_test(NAME cli_version COMMAND wtop --version)
        add_test(NAME cli_capabilities COMMAND wtop --capabilities --format json)
        add_test(NAME cli_gui_reserved COMMAND wtop --gui)
        set_tests_properties(cli_gui_reserved PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
        add_test(NAME cli_snapshot COMMAND wtop --snapshot --format json --interval 50 --no-config)
        set_tests_properties(cli_version cli_capabilities PROPERTIES TIMEOUT 5)
        set_tests_properties(cli_snapshot PROPERTIES TIMEOUT 15)
        find_package(Python3 COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            add_test(NAME native_cli_behavior
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_cli.py" "$<TARGET_FILE:wtop>")
            set_tests_properties(native_cli_behavior PROPERTIES TIMEOUT 90)
            add_test(NAME native_provider_behavior
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_providers.py" "$<TARGET_FILE:wtop>")
            set_tests_properties(native_provider_behavior PROPERTIES TIMEOUT 90)
            add_test(NAME native_tui_interaction
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_tui.py" "$<TARGET_FILE:wtop>")
            set_tests_properties(native_tui_interaction PROPERTIES TIMEOUT 180)
        endif()
    endif()
else()
    message(STATUS "Building portable core tests only; the current application backend requires Windows.")
endif()
