cmake_minimum_required(VERSION 3.16)
project(Wtop LANGUAGES C)
include(CTest)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" WTOP_VERSION LIMIT_COUNT 1)
configure_file(source/version.h.in generated/version.h @ONLY)
add_library(wtop_core STATIC source/core.c source/settings.c source/ui.c)
target_include_directories(wtop_core PUBLIC source)
set_target_properties(wtop_core PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(BUILD_TESTING)
    foreach(component IN ITEMS core settings ui)
        add_executable(test_${component} tests/test_${component}.c)
        target_link_libraries(test_${component} PRIVATE wtop_core)
        set_target_properties(test_${component} PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if(MSVC)
            target_compile_options(test_${component} PRIVATE /UNDEBUG)
        else()
            target_compile_options(test_${component} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME portable_${component} COMMAND test_${component})
    endforeach()
endif()
if(WIN32)
    add_executable(wtop source/ntop.c source/util.c source/vi.c)
    target_include_directories(wtop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(wtop PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 PSAPI_VERSION=1)
    target_link_libraries(wtop PRIVATE wtop_core advapi32 user32 psapi)
    if(MSVC)
        target_compile_options(wtop PRIVATE /W4)
        set_property(TARGET wtop wtop_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        if(BUILD_TESTING)
            set_property(TARGET test_core test_settings test_ui PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
    elseif(MINGW)
        target_link_options(wtop PRIVATE -municode)
        target_link_libraries(wtop PRIVATE m)
    endif()
    if(BUILD_TESTING)
        add_test(NAME cli_version COMMAND wtop --version)
        add_test(NAME cli_capabilities COMMAND wtop --capabilities --format json)
        add_test(NAME cli_gui_reserved COMMAND wtop --gui)
        set_tests_properties(cli_gui_reserved PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
        add_test(NAME cli_snapshot COMMAND wtop --snapshot --format json --interval 50 --no-config)
        set_tests_properties(cli_version cli_capabilities PROPERTIES TIMEOUT 5)
        set_tests_properties(cli_snapshot PROPERTIES TIMEOUT 15)
        find_package(Python3 COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            add_test(NAME native_cli_behavior
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_cli.py" "$<TARGET_FILE:wtop>")
            set_tests_properties(native_cli_behavior PROPERTIES TIMEOUT 90)
            add_test(NAME native_tui_interaction
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_tui.py" "$<TARGET_FILE:wtop>")
            set_tests_properties(native_tui_interaction PROPERTIES TIMEOUT 90)
        endif()
    endif()
else()
    message(STATUS "Building portable core tests only; the current application backend requires Windows.")
endif()
