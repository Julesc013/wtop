cmake_minimum_required(VERSION 3.16)
project(WtopInstallerGuard LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "The embedded MSI guard requires a Windows target")
endif()
if(NOT EXISTS "${WTOP_GUARD_CONFIG_DIR}/guard_config.h")
    message(FATAL_ERROR "Run packaging/build_msi.py to bind the guard to exact payload bytes")
endif()
add_library(wtop_installer_guard SHARED guard.c)
target_include_directories(wtop_installer_guard PRIVATE "${WTOP_GUARD_CONFIG_DIR}")
target_compile_definitions(wtop_installer_guard PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601)
target_link_libraries(wtop_installer_guard PRIVATE msi advapi32 shell32 rstrtmgr)
set_target_properties(wtop_installer_guard PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO PREFIX "")
if(MSVC)
    target_compile_options(wtop_installer_guard PRIVATE /W4 /WX)
    set_property(TARGET wtop_installer_guard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/guard.def" "EXPORTS\nWtopGuard=_WtopGuard@4\nWtopGuardDeferred=_WtopGuardDeferred@4\n")
    else()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/guard.def" "EXPORTS\nWtopGuard\nWtopGuardDeferred\n")
    endif()
    target_sources(wtop_installer_guard PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/guard.def")
else()
    target_compile_options(wtop_installer_guard PRIVATE -Wall -Wextra -Werror)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4 AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
        target_link_options(wtop_installer_guard PRIVATE -Wl,--kill-at)
    endif()
endif()
